// CONFIGURATION GÉOGRAPHIQUE DE L'ÉCOLE (MARLOIE) const LAT = 50.201; const LON = 5.313; // CALCUL AUTOMATIQUE DU CYCLE DE LA LUNE function calculerPhaseLune(date) { let lp = 2551443; let nouvelleLuneRef = new Date(1970, 0, 7, 20, 35, 0); let phase = ((date.getTime() - nouvelleLuneRef.getTime()) / 1000) % lp; let age = phase / (24 * 3600); let nomPhase = "Pleine lune"; let iconePhase = "🌕"; let illumination = 100; if (age < 1.84) { nomPhase = "Nouvelle Lune"; iconePhase = "🌑"; illumination = 0; } else if (age < 5.53) { nomPhase = "Premier Croissant"; iconePhase = "🌒"; illumination = 25; } else if (age < 9.22) { nomPhase = "Premier Quartier"; iconePhase = "🌓"; illumination = 50; } else if (age < 12.91) { nomPhase = "Gibbeuse Croissante"; iconePhase = "🌔"; illumination = 75; } else if (age < 16.61) { nomPhase = "Pleine Lune"; iconePhase = "🌕"; illumination = 100; } else if (age < 20.30) { nomPhase = "Gibbeuse Décroissante"; iconePhase = "🌖"; illumination = 75; } else if (age < 23.99) { nomPhase = "Dernier Quartier"; illumination = 50; } else if (age < 27.68) { nomPhase = "Dernier Croissant"; illumination = 25; } else { nomPhase = "Nouvelle Lune"; illumination = 0; } document.getElementById('sat-lune-icone').textContent = iconePhase; document.querySelector('#sat-lune-icone').parentNode.querySelector('strong').textContent = nomPhase; document.querySelector('#sat-lune-icone').parentNode.innerHTML = `
Lune
${nomPhase} Âge : ${age.toFixed(1)} jours
Éclairement : ${illumination}%
${iconePhase}
`; } // CALCUL DE LA SAISON ET ÉVOLUTION DE LA GRAINE function calculerSaison(date) { const mois = date.getMonth() + 1; const jour = date.getDate(); let nomSaison = "ÉTÉ"; let iconeGraine = "🌳"; if ((mois == 3 && jour >= 21) || mois == 4 || mois == 5 || (mois == 6 && jour < 21)) { nomSaison = "PRINTEMPS"; iconeGraine = "🌱"; } else if ((mois == 6 && jour >= 21) || mois == 7 || mois == 8 || (mois == 9 && jour < 21)) { nomSaison = "ÉTÉ"; iconeGraine = "🌳"; } else if ((mois == 9 && jour >= 21) || mois == 10 || mois == 11 || (mois == 12 && jour < 21)) { nomSaison = "AUTOMNE"; iconeGraine = "🍂"; } else { nomSaison = "HIVER"; iconeGraine = "❄️"; } document.getElementById('sat-saison').textContent = nomSaison; document.getElementById('sat-saison').nextElementSibling.textContent = iconeGraine; } // APPEL DE L'API POUR DEMANDER LA METEO EN DIRECT async function recupererMeteo() { try { const url = `https://open-meteo.com{LAT}&longitude=${LON}¤t=temperature_2m,relative_humidity_2m,apparent_temperature,weather_code,wind_speed_10m,uv_index&daily=weather_code,temperature_2m_max,temperature_2m_min,sunrise,sunset&timezone=Europe%2FBrussels`; const reponse = await fetch(url); const donnees = await reponse.json(); // Injection météo actuelle const c = donnees.current; document.getElementById('courant-temp').textContent = `${Math.round(c.temperature_2m)}°C`; document.getElementById('courant-ressenti').textContent = `${Math.round(c.apparent_temperature)}°C`; document.getElementById('courant-humidite').textContent = `${c.relative_humidity_2m}%`; document.getElementById('courant-vent').textContent = `${Math.round(c.wind_speed_10m)} km/h`; document.getElementById('courant-uv').textContent = `${c.uv_index} / ${c.uv_index <= 2 ? 'Faible' : 'Modéré'}`; // Traduction du code météo analyserCodeMeteo(c.weather_code); // Éphémérides Soleil document.getElementById('sat-lever').textContent = donnees.daily.sunrise[0].split('T')[1]; document.getElementById('sat-coucher').textContent = donnees.daily.sunset[0].split('T')[1]; // Appel de la mise à jour Lune et Saison calculerPhaseLune(new Date()); calculerSaison(new Date()); } catch (e) { console.error("Erreur API Météo:", e); } } // TRADUCTION DE VOTRE LOGIQUE DE CODES MÉTÉO (CONFORME À VOTRE CODE D'ORIGINE) function analyserCodeMeteo(code) { let icone = "☁️"; let description = "Couvert"; const fond = document.getElementById('fond-ecran-principal'); if (code <= 1) { icone = "☀️"; description = "Ensoleillé"; fond.style.filter = "brightness(1.0) contrast(1)"; } else if ([2, 3, 45, 48].includes(code)) { icone = "☁️"; description = "Nuageux"; fond.style.filter = "brightness(0.85) contrast(0.95)"; } else if ([51, 53, 55, 61, 63, 65, 80, 81, 82].includes(code)) { icone = "🌧️"; description = "Pluvieux"; fond.style.filter = "brightness(0.6) contrast(0.9)"; } else if ([71, 73, 75, 77, 85, 86].includes(code)) { icone = "❄️"; description = "Neige"; fond.style.filter = "brightness(0.75) contrast(1.05)"; } else if ([95, 96, 99].includes(code)) { icone = "⛈️"; description = "Orageux"; fond.style.filter = "brightness(0.5) contrast(1.1)"; } document.getElementById('courant-icone').textContent = icone; document.getElementById('courant-desc').textContent = description; } // FONCTION DE TRADUCTION POUR LES AVANTS-PREVISIONS DU BAS function getIconByCode(code) { if (code <= 1) return "☀️"; if ([2, 3, 45, 48].includes(code)) return "☁️"; if ([51, 53, 55, 61, 63, 65, 80, 81, 82].includes(code)) return "🌧️"; if ([71, 73, 75, 77, 85, 86].includes(code)) return "❄️"; return "⛈️"; } // INITIATION ET RAFRAÎCHISSEMENT TOUTES LES 5 MINUTES recupererMeteo(); setInterval(recupererMeteo, 300000);
